                <h2 id="overlayTitle" class="text-lg md:text-2xl text-emerald-400 text-glow mb-4 uppercase">READY PLAYER ONE</h2>
                <p id="overlayDesc" class="text-[10px] md:text-xs text-slate-300 mb-6 leading-relaxed">Use Arrow Keys or On-Screen buttons to guide the snake. Eat food, avoid walls and your own tail!</p>
                <button id="actionBtn" class="bg-emerald-600 hover:bg-emerald-500 text-black font-bold px-6 py-3 rounded-lg text-xs md:text-sm btn-arcade uppercase tracking-wider">Start Game</button>

            <button id="upBtn" class="bg-slate-800 hover:bg-slate-700 border border-emerald-500/50 rounded-lg flex items-center justify-center text-emerald-400 text-lg active:bg-emerald-600 active:text-black">▲</button>

            <button id="downBtn" class="bg-slate-800 hover:bg-slate-700 border border-emerald-500/50 rounded-lg flex items-center justify-center text-emerald-400 text-lg active:bg-emerald-600 active:text-black">▼</button>
            <button id="rightBtn" class="bg-slate-800 hover:bg-slate-700 border border-emerald-500/50 rounded-lg flex items-center justify-center text-emerald-400 text-lg active:bg-emerald-600 active:text-black">▶</button>

    <script>
        const canvas = document.getElementById('gameCanvas');
        const ctx = canvas.getContext('2d');
        const scoreEl = document.getElementById('score');
        const highScoreEl = document.getElementById('highScore');
        const overlay = document.getElementById('overlay');
        const overlayTitle = document.getElementById('overlayTitle');
        const overlayDesc = document.getElementById('overlayDesc');
        const actionBtn = document.getElementById('actionBtn');

        // Grid dimensions
        const GRID_SIZE = 20;
        let tileCount = 20;
        let tileSize = canvas.width / tileCount;

        // Game variables
        let snake = [];
        let food = { x: 0, y: 0 };
        let dx = 1;
        let dy = 0;
        let score = 0;
        let highScore = localStorage.getItem('snakeHighScore') || 0;
        highScoreEl.textContent = highScore;

        let gameInterval = null;
        let gameSpeed = 100; // ms per tick
        let gameState = 'START'; // START, PLAYING, PAUSED, GAMEOVER
        let particles = [];

        // Resize canvas correctly
        function resizeCanvas() {
            const container = canvas.parentElement;
            canvas.width = container.clientWidth;
            canvas.height = container.clientHeight;
            tileSize = canvas.width / tileCount;
            draw();
        }
        window.addEventListener('resize', resizeCanvas);

        // Initialize game state
        function initGame() {
            snake = [
                { x: 10, y: 10 },
                { x: 9, y: 10 },
                { x: 8, y: 10 }
            ];
            dx = 1;
            dy = 0;
            score = 0;
            scoreEl.textContent = score;
            gameSpeed = 100;
            spawnFood();
            particles = [];
        }

        // Spawn food on grid avoiding snake
        function spawnFood() {
            let valid = false;
            while (!valid) {
                food.x = Math.floor(Math.random() * tileCount);
                food.y = Math.floor(Math.random() * tileCount);
                valid = !snake.some(segment => segment.x === food.x && segment.y === food.y);
            }
        }

        // Start game handler
        function startGame() {
            initGame();
            gameState = 'PLAYING';
            overlay.classList.add('opacity-0', 'pointer-events-none');
            if (gameInterval) clearInterval(gameInterval);
            gameInterval = setInterval(gameLoop, gameSpeed);
        }

        // Main game loop
        function gameLoop() {
            update();
            draw();
        }

        // Update positions & collision check
        function update() {
            if (gameState !== 'PLAYING') return;

            // Calculate new head position
            const head = { x: snake[0].x + dx, y: snake[0].y + dy };

            // Wall collision check
            if (head.x < 0 || head.x >= tileCount || head.y < 0 || head.y >= tileCount) {
                gameOver();
                return;
            }

            // Self collision check
            if (snake.some(segment => segment.x === head.x && segment.y === head.y)) {
                gameOver();
                return;
            }

            snake.unshift(head);

            // Eat food check
            if (head.x === food.x && head.y === food.y) {
                score += 10;
                scoreEl.textContent = score;
                if (score > highScore) {
                    highScore = score;
                    highScoreEl.textContent = highScore;
                    localStorage.setItem('snakeHighScore', highScore);
                }
                createParticles(food.x, food.y);
                spawnFood();

                // Increase speed slightly
                if (gameSpeed > 50) {
                    gameSpeed -= 2;
                    clearInterval(gameInterval);
                    gameInterval = setInterval(gameLoop, gameSpeed);
                }
            } else {
                snake.pop();
            }

            // Update particle effects
            particles.forEach((p, index) => {
                p.life--;
                p.x += p.vx;
                p.y += p.vy;
                if (p.life <= 0) particles.splice(index, 1);
            });
        }

        // Create food particles
        function createParticles(gx, gy) {
            const px = (gx + 0.5) * tileSize;
            const py = (gy + 0.5) * tileSize;
            for (let i = 0; i < 12; i++) {
                const angle = Math.random() * Math.PI * 2;
                const speed = Math.random() * 3 + 1;
                particles.push({
                    x: px,
                    y: py,
                    vx: Math.cos(angle) * speed,
                    vy: Math.sin(angle) * speed,
                    life: 15,
                    color: '#34d399'
                });
            }
        }

        // Render game graphics
        function draw() {
            ctx.fillStyle = '#050505';
            ctx.fillRect(0, 0, canvas.width, canvas.height);

            // Draw grid lines subtly
            ctx.strokeStyle = '#0f172a';
            ctx.lineWidth = 0.5;
            for (let i = 0; i < tileCount; i++) {
                ctx.beginPath();
                ctx.moveTo(i * tileSize, 0);
                ctx.lineTo(i * tileSize, canvas.height);
                ctx.stroke();

                ctx.beginPath();
                ctx.moveTo(0, i * tileSize);
                ctx.lineTo(canvas.width, i * tileSize);
                ctx.stroke();
            }

            // Draw food with glow
            ctx.shadowBlur = 12;
            ctx.shadowColor = '#f87171';
            ctx.fillStyle = '#f87171';
            ctx.beginPath();
            ctx.arc((food.x + 0.5) * tileSize, (food.y + 0.5) * tileSize, tileSize * 0.4, 0, Math.PI * 2);
            ctx.fill();
            ctx.shadowBlur = 0; // reset

            // Draw snake
            snake.forEach((segment, index) => {
                ctx.fillStyle = index === 0 ? '#34d399' : '#059669';
                ctx.shadowBlur = index === 0 ? 10 : 0;
                ctx.shadowColor = '#34d399';
                ctx.fillRect(segment.x * tileSize + 1, segment.y * tileSize + 1, tileSize - 2, tileSize - 2);
            });
            ctx.shadowBlur = 0;

            // Draw particles
            particles.forEach(p => {
                ctx.fillStyle = p.color;
                ctx.fillRect(p.x, p.y, 3, 3);
            });
        }

        // Game over handler
        function gameOver() {
            gameState = 'GAMEOVER';
            clearInterval(gameInterval);
            overlayTitle.textContent = "GAME OVER";
            overlayDesc.textContent = `Final Score: ${score}\nGreat effort! Ready to beat your high score?`;
            actionBtn.textContent = "Play Again";
            overlay.classList.remove('opacity-0', 'pointer-events-none');
        }

        // Keyboard controls
        window.addEventListener('keydown', e => {
            if (['ArrowUp', 'ArrowDown', 'ArrowLeft', 'ArrowRight', ' '].includes(e.key)) {
                e.preventDefault();
            }
            if (gameState !== 'PLAYING') return;

            switch (e.key) {
                case 'ArrowUp':
                    if (dy === 0) { dx = 0; dy = -1; }
                    break;
                case 'ArrowDown':
                    if (dy === 0) { dx = 0; dy = 1; }
                    break;
                case 'ArrowLeft':
                    if (dx === 0) { dx = -1; dy = 0; }
                    break;
                case 'ArrowRight':
                    if (dx === 0) { dx = 1; dy = 0; }
                    break;
            }
        });

        // Touch control buttons
        document.getElementById('upBtn').addEventListener('click', () => { if (gameState === 'PLAYING' && dy === 0) { dx = 0; dy = -1; } });
        document.getElementById('downBtn').addEventListener('click', () => { if (gameState === 'PLAYING' && dy === 0) { dx = 0; dy = 1; } });
        document.getElementById('leftBtn').addEventListener('click', () => { if (gameState === 'PLAYING' && dx === 0) { dx = -1; dy = 0; } });
        document.getElementById('rightBtn').addEventListener('click', () => { if (gameState === 'PLAYING' && dx === 0) { dx = 1; dy = 0; } });

        actionBtn.addEventListener('click', startGame);

        // Initial setup
        resizeCanvas();
        draw();
    </script>
